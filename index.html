How you'd use it

# tracker.py — a slice of the shape; scheduler, digest & alert code in the full script
tracker = PriceTracker(delay_s=2.0, user_agent='pricebot/1.0 (+you@example.com)')
tracker.watch('https://rival-a.example.com/headphones-x200',
              selector='span.price--main', alert_drop_pct=3.0)
tracker.watch('https://rival-b.example.com/hp-x200',
              selector="meta[property='product:price:amount]'")

tracker.run(schedule='daily@09:00')  # → history.csv, history.json, alerts
